function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Fetch Items var itemsSheet = ss.getSheetByName("Items"); var itemsData = itemsSheet ? itemsSheet.getDataRange().getDisplayValues() : []; var items = []; for (var i = 1; i < itemsData.length; i++) { if (itemsData[i][0] !== "") { items.push({ id: String(itemsData[i][0]).trim(), name: itemsData[i][1], storage: Number(itemsData[i][2]) || 0, display: Number(itemsData[i][3]) || 0 }); } } // 2. Fetch Borrows using getDisplayValues (returns text directly, no Date objects) var borrowsSheet = ss.getSheetByName("Borrows"); var borrowsData = borrowsSheet ? borrowsSheet.getDataRange().getDisplayValues() : []; var borrows = []; for (var j = 1; j < borrowsData.length; j++) { if (borrowsData[j][0] !== "") { borrows.push({ id: String(borrowsData[j][0]).trim(), itemId: String(borrowsData[j][1]).trim(), borrower: String(borrowsData[j][2]).trim(), qty: Number(borrowsData[j][3]) || 0, borrowDate: String(borrowsData[j][4]).trim(), returnDate: String(borrowsData[j][5]).trim(), status: String(borrowsData[j][6]).trim(), contact: String(borrowsData[j][7] || "").trim() }); } } var output = JSON.stringify({ items: items, borrows: borrows }); return ContentService.createTextOutput(output) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = {}; try { data = JSON.parse(e.postData.contents); } catch (err) { if (e.parameter && e.parameter.payload) { data = JSON.parse(e.parameter.payload); } } if (data.action === "ADD_BORROW") { var borrowsSheet = ss.getSheetByName("Borrows"); borrowsSheet.appendRow([ "B-" + Date.now(), data.itemId, data.borrower, data.qty, data.borrowDate, data.returnDate, "BORROWED", "'" + data.contact ]); } else if (data.action === "RETURN_ITEM") { var borrowsSheet = ss.getSheetByName("Borrows"); var borrowsData = borrowsSheet.getDataRange().getValues(); for (var k = 1; k < borrowsData.length; k++) { if (String(borrowsData[k][0]).trim() === String(data.borrowId).trim()) { borrowsSheet.getRange(k + 1, 7).setValue("RETURNED"); break; } } } else if (data.action === "UPDATE_STOCK") { var logsSheet = ss.getSheetByName("Logs"); logsSheet.appendRow([ "L-" + Date.now(), data.itemId, data.type, data.qty, data.location, new Date().toISOString().split('T')[0], data.notes ]); var itemsSheet = ss.getSheetByName("Items"); var itemsData = itemsSheet.getDataRange().getValues(); for (var m = 1; m < itemsData.length; m++) { if (String(itemsData[m][0]).trim() === String(data.itemId).trim()) { var colIndex = (data.location === "Storage") ? 3 : 4; var currentQty = Number(itemsData[m][colIndex - 1]) || 0; var change = Number(data.qty); var newQty = (data.type === "ADD") ? currentQty + change : currentQty - change; if (newQty < 0) newQty = 0; itemsSheet.getRange(m + 1, colIndex).setValue(newQty); break; } } } return ContentService.createTextOutput(JSON.stringify({ result: "success" })) .setMimeType(ContentService.MimeType.JSON); }
